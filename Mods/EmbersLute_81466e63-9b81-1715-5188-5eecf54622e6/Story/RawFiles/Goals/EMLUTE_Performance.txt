// Adding some restrictions to enforce only one lead conductor at a time
// Aim is to improve stability by preventing scenarios that BG3 likes to CTD about
//REGION music performance range tracker
IF
TagSet((CHARACTER)_Performer,(TAG)EMLUTE_HAS_ANY_AURA_3623a4cd-f630-4ba8-9ca2-0914d345802c)
THEN
ClearTag(_Performer, (TAG)EMLUTE_PARTY_HAS_LEAD_PERFORMER_10b86b2c-8cdf-44fc-a920-a7c5562faf6d);
PROC_TagRestOfPartyForLeadPerformer(_Performer);

IF
TagCleared((CHARACTER)_Performer,(TAG)EMLUTE_HAS_ANY_AURA_3623a4cd-f630-4ba8-9ca2-0914d345802c)
THEN
ClearTag(_Performer, (TAG)EMLUTE_PARTY_HAS_LEAD_PERFORMER_10b86b2c-8cdf-44fc-a920-a7c5562faf6d);
PROC_TagRemoveRestOfPartyForLeadPerformer(_Performer);

PROC
PROC_TagRestOfPartyForLeadPerformer((CHARACTER)_Performer)
AND
DB_PartOfTheTeam(_PartyMember)
AND
_PartyMember != _Performer
AND
IsTagged(_PartyMember,(TAG)EMLUTE_PARTY_HAS_LEAD_PERFORMER_10b86b2c-8cdf-44fc-a920-a7c5562faf6d,0)
THEN
SetTag(_PartyMember, (TAG)EMLUTE_PARTY_HAS_LEAD_PERFORMER_10b86b2c-8cdf-44fc-a920-a7c5562faf6d);

PROC
PROC_TagRemoveRestOfPartyForLeadPerformer((CHARACTER)_Performer)
AND
DB_PartOfTheTeam(_PartyMember)
AND
_PartyMember != _Performer
AND
IsTagged(_PartyMember,(TAG)EMLUTE_PARTY_HAS_LEAD_PERFORMER_10b86b2c-8cdf-44fc-a920-a7c5562faf6d,1)
THEN
ClearTag(_PartyMember, (TAG)EMLUTE_PARTY_HAS_LEAD_PERFORMER_10b86b2c-8cdf-44fc-a920-a7c5562faf6d);
//END_REGION

// This exists because auras suck and I'm bored of trying to stop them tripping CTDs
// Maybe at some point I nuke all the auras and replace with Osiris
//REGION music combat support
IF
TagSet((CHARACTER)_Performer,(TAG)EMLUTE_COMBAT_SUPPORT_ACTIVE_d8cc0958-1777-4a4a-97d8-e0ca48cdfeee)
AND
GetEquippedItem(_Performer,"MusicalInstrument",_Instrument)
AND
DB_InstrumentVFX_ToCombatSupportStatus_EmbersLute(_Tag,_Status)
AND
IsTagged(_Instrument,_Tag,1)
THEN
PROC_ApplyCombatBuffToParty_EmbersLute(_Status,_Performer);

PROC
PROC_ApplyCombatBuffToParty_EmbersLute((STRING)_Status, (CHARACTER)_Cause)
AND
DB_Players(_Player)
AND
IsTagged(_Player,(TAG)EMLUTE_HAS_ANY_AURA_3623a4cd-f630-4ba8-9ca2-0914d345802c,0)
AND
HasActiveStatus(_Player,_Status,0)
THEN
ApplyStatus(_Player,_Status,-1.0,0,_Cause);

IF
TagCleared((CHARACTER)_Performer,(TAG)EMLUTE_COMBAT_SUPPORT_ACTIVE_d8cc0958-1777-4a4a-97d8-e0ca48cdfeee)
AND
DB_InstrumentVFX_ToCombatSupportStatus_EmbersLute(_Tag,_Status)
THEN
PROC_RemoveCombatBuffFromParty_EmbersLute(_Status);

PROC
PROC_RemoveCombatBuffFromParty_EmbersLute((STRING)_Status)
AND
DB_Players(_Player)
AND
HasActiveStatus(_Player,_Status,0)
THEN
RemoveStatus(_Player,_Status);

//END_REGION

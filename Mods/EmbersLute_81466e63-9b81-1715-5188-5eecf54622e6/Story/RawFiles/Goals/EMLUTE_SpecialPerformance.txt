// Starts music on the lead performer, caches active performance and begins a timer
// Also throws a couple of helper items onto the ground
PROC
PROC_StartSpecialPerformance_EmbersLute((CHARACTER)_Performer,(STRING)_Song,(STRING)_StartEvent,(INTEGER)_Duration)
AND
IsTagged(_Performer, (TAG)EMLUTE_LEGEND_LEGACY_MODE_15462095-925f-4ec3-853f-8cae55d93904,0)
AND
NOT DB_ActiveSpecialPerformance_EmbersLute(_Performer,_Song,_)
AND
DB_SpecialPerformance_Helper_EmbersLute(_HelperTemplate)
AND
CreateAtObject(_HelperTemplate,_Performer,0,0,"",1,(ITEM)_HelperItem)
AND
GetPosition(_Performer,_X,_Y,_Z)
AND
RealSum(_Z,-0.5,_OffsetZ)
THEN
TeleportToPosition(_HelperItem,_X,_Y,_OffsetZ,"",0,0,0,0,1);
ObjectTimerCancel(_Performer,_Song);
RealtimeObjectTimerLaunch(_Performer,_Song,_Duration);
PROC_PlayMusicOnItem(_Performer,_Song,_HelperItem);
DB_SpecialPerformance_HelperItemAudit_EmbersLute(_Song,_HelperItem);

PROC
PROC_PlayMusicOnItem((CHARACTER)_Performer,(STRING)_Song,(ITEM)_HelperItem)
THEN
DB_ActiveSpecialPerformance_EmbersLute((CHARACTER)_Performer,(STRING)_Song,(ITEM)_HelperItem);
RealtimeObjectTimerLaunch(_HelperItem, "PlayMusicOnItem_1_Timer_Emlute",100);

// Forcing the sound events in sequence, as it seems to matter.
// Using timer to chain, since not sure if PlaySoundResource is immediate, or if async could mess up sequence.
IF
ObjectTimerFinished((ITEM)_HelperItem, "PlayMusicOnItem_1_Timer_Emlute")
AND
DB_ActiveSpecialPerformance_EmbersLute(_Performer,_Song,_HelperItem)
AND 
DB_SpecialPerformance_Mapping_EmbersLute(_Song,_MusicSoundResource,1)
THEN
PlaySoundResource(_HelperItem,_MusicSoundResource);
RealtimeObjectTimerLaunch(_HelperItem, "PlayMusicOnItem_2_Timer_Emlute",100);
IF
ObjectTimerFinished((ITEM)_HelperItem, "PlayMusicOnItem_2_Timer_Emlute")
AND
DB_ActiveSpecialPerformance_EmbersLute(_Performer,_Song,_HelperItem)
AND 
DB_SpecialPerformance_Mapping_EmbersLute(_Song,_MusicSoundResource,2)
THEN
PlaySoundResource(_HelperItem,_MusicSoundResource);
RealtimeObjectTimerLaunch(_HelperItem, "PlayMusicOnItem_3_Timer_Emlute",100);
IF
ObjectTimerFinished((ITEM)_HelperItem, "PlayMusicOnItem_3_Timer_Emlute")
AND
DB_ActiveSpecialPerformance_EmbersLute(_Performer,_Song,_HelperItem)
AND 
DB_SpecialPerformance_Mapping_EmbersLute(_Song,_MusicSoundResource,3)
THEN
PlaySoundResource(_HelperItem,_MusicSoundResource);
RealtimeObjectTimerLaunch(_HelperItem, "PlayMusicOnItem_4_Timer_Emlute",100);
IF
ObjectTimerFinished((ITEM)_HelperItem, "PlayMusicOnItem_4_Timer_Emlute")
AND
DB_ActiveSpecialPerformance_EmbersLute(_Performer,_Song,_HelperItem)
AND 
DB_SpecialPerformance_Mapping_EmbersLute(_Song,_MusicSoundResource,4)
THEN
PlaySoundResource(_HelperItem,_MusicSoundResource);
IF
ObjectTimerFinished((CHARACTER)_Performer, "IncreaseVolume_Timer_Emlute")
AND
DB_ActiveSpecialPerformance_EmbersLute(_Performer,_,_)
THEN
MusicPlayGeneral("UI_State_NonHUDWidget_On");

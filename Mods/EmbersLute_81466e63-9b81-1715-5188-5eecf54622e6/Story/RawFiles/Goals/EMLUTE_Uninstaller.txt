KBSECTION
// We loop through all player characters, find any custom items and unequip.
// The timer procs every half a second to check when all custom items have been unequipped.
// Once nothing is found, it then deletes all the items with the custom mod tag.

// n.b. this won't work on items scattered across the realm
// We'll recommend the player takes a long rest to clean up, before attempting to uninstall.

IF
CastedSpell((CHARACTER)_Caster,"Shout_CleanUp_Emlute",_,_,_)
AND
DB_Players(_Player)
THEN
OpenMessageBoxYesNo(_Caster,"EMLUTE_ConfirmUninstall");

IF
MessageBoxYesNoClosed(_Player,"EMLUTE_ConfirmUninstall",1)
THEN
OpenMessageBoxYesNo(_Player,"EMLUTE_ConfirmUninstall_2");

IF
MessageBoxYesNoClosed(_Player,"EMLUTE_ConfirmUninstall_2",1)
THEN
OpenMessageBoxYesNo(_Player,"EMLUTE_ConfirmUninstall_3");

IF
MessageBoxYesNoClosed(_Player,"EMLUTE_ConfirmUninstall_3",1)
THEN
DB_ItemCleanUp_EmbersLute(_Player);
PROC_UnequipCustom_EmbersLute();
TimerLaunch("DeleteItem_EmbersLute",1500);

// No more equipped items found, so safe to delete now
IF
TimerFinished("DeleteItem_EmbersLute")
AND
NOT QRY_AtLeastOneCustomItemStillEquipped_EmbersLute()
AND
DB_ItemCleanUp_EmbersLute(_Player)
THEN
PROC_RemoveCustomSpells_EmbersLute();
PROC_RemoveCustomPassives_EmbersLute();
PROC_RemoveCustomStatuses_EmbersLute();
//PROC_CleanUpVanillaTables_EmbersLute();
PROC_CleanUpCustom_EmbersLute();
PROC_ClearPerformingNPCBards_EmbersLute();
PROC_CleanUpFailsafe_EmbersLute();
PROC_ScriptedVFX_CleanUp_Failsafe_EmbersLute();
PROC_ResetRewardTracker_EmbersLute();
OpenMessageBox(_Player,"EMLUTE_CompletedUninstall");
NOT DB_ItemCleanUp_EmbersLute(_Player);

IF
MessageBoxClosed(_Player,"EMLUTE_CompletedUninstall")
THEN
OpenMessageBox(_Player,"EMLUTE_CompletedUninstall_2");

// Put a failsafe in for max 10 wait loops
IF
TimerFinished("DeleteItem_EmbersLute")
AND
DB_IterateDeleteCounter_EmbersLute(_Count)
AND
_Count < 10
AND
QRY_AtLeastOneCustomItemStillEquipped_EmbersLute()
AND
IntegerSum(_Count,1,_CountIncrement)
THEN
DB_IterateDeleteCounter_EmbersLute(_CountIncrement);
TimerLaunch("DeleteItem_EmbersLute",500);

IF
EntityEvent((ITEM)_Item, "EMLUTE_FoundEmluteItem")
AND
GetTemplate(_Item,(ITEMROOT)_Template)
AND
DB_ContainerTemplates_EmbersLute(_Template)
THEN
IterateInventoryByTag(_Item,"EMLUTE_a56f1251-c4f5-4570-97a1-d0bd7be19086","EMLUTE_FoundEmluteItem","EMLUTE_ContainerSearchDone");

IF
EntityEvent((ITEM)_Item, "EMLUTE_FoundEmluteItem")
AND
GetTemplate(_Item,(ITEMROOT)_Template)
AND
NOT QRY_IsContainer_EmbersLute(_Template)
THEN
RequestDelete(_Item);

QRY
QRY_IsContainer_EmbersLute((ITEMROOT)_Template)
AND
DB_ContainerTemplates_EmbersLute(_Template)
THEN
DB_NOOP(1);

IF
EntityEvent((ITEM)_Container, "EMLUTE_ContainerSearchDone")
THEN
RequestDelete(_Container);

IF
EntityEvent((ITEM)_Item, "EMLUTE_PlayerSearchDone")
THEN
DB_NOOP(1);

IF
EntityEvent((ITEM)_Item, "EMLUTE_CampSearchDone")
THEN
DB_NOOP(1);

IF
EntityEvent((ITEM)_Item, "EMLUTE_TraderSearchDone")
THEN
DB_NOOP(1);

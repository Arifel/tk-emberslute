//REGION 
// Here we are overloading vanilla so it handles our custom disturbance too.
// Want to re-use the flourishes, so hook onto the same timer event thrown.
IF
OnCrimeInvestigatorSwitchedState(_ID,_NPC,_,"Confronting")
AND
DB_CRIME_MusicalPerformance(_ID,_Performer,"MusicalPerformance_Good_Emlute")
AND
NOT DB_CRIME_MusicalPerformance_NPCs(_ID,_Performer,_)
THEN
PROC_CRIME_MusicalPerformance_Good_LaunchTimer(_Performer);

// Vanilla overload for our custom disturbance.
IF
OnCrimeInvestigatorSwitchedState(_ID,_NPC,_,"Confronting")
AND
DB_CRIME_MusicalPerformance(_ID,_Performer,"MusicalPerformance_Good_Emlute")
AND
NOT DB_CRIME_MusicalPerformance_NPCs(_ID,_Performer,_NPC)
THEN
DB_CRIME_MusicalPerformance_NPCs(_ID,_Performer,_NPC);
PROC_ObjectCountHelper(_Performer,"CRIME_MusicalPerformance_NPCs");

// Vanilla overload for our custom disturbance.
// DC based on number of reacting NPCs
IF
ObjectTimerFinished(_Performer,"CRIME_MusicalPeformance_Good")
AND
DB_CRIME_MusicalPerformance(_ID,(CHARACTER)_Performer,"MusicalPerformance_Good_Emlute")
AND
DB_ObjectCountHelper(_Performer,"CRIME_MusicalPerformance_NPCs",_NPCCount)
AND
QRY_OnlyOnce_Reset("CRIME_MusicalPerformance_DC")
AND
DB_CRIME_MusicalPerformance_DC(_MinNPC,_DC)
AND
_NPCCount >= _MinNPC
AND
QRY_OnlyOnce("CRIME_MusicalPerformance_DC")
THEN
RequestPassiveRoll((CHARACTER)_Performer,_Performer,"SkillCheck","Performance",_DC,0,"CRIME_MusicalPerformance_PerformanceCheck");

// Vanilla overload for our custom disturbance.
PROC
PROC_CRIME_Finished((INTEGER)_ID)
AND
DB_CRIME_MusicalPerformance(_ID,_Performer,"MusicalPerformance_Good_Emlute")
THEN
ObjectTimerCancel(_Performer,"CRIME_MusicalPeformance_Good");
ObjectTimerCancel(_Performer,"EMLUTE_BardReactions_Flourish_Tick");
PROC_ObjectCountHelperReset(_Performer,"CRIME_MusicalPerformance_NPCs");
//END_REGION

PROC_CacheCooldownState_EmbersLute(_Charges);
PROC_DoFlourishNPC_EmbersLute(_Performer);

// Not sure that entity events always fire on anims?
// Remove suspect NPCs still in cooldown after 2 ticks
PROC

AND
DB_BardReactionAnims_CooldownCache_Emlute(_Charge1,_NPC)
AND

AND
_Charge1 != _Charge2

NOT DB_BardReactionAnims_Cooldown_Emlute(_NPC);
NOT DB_BardReactionAnims_CooldownCache_Emlute(_Charge1,_NPC);
NOT DB_BardReactionAnims_CooldownCache_Emlute(_Charge2,_NPC);

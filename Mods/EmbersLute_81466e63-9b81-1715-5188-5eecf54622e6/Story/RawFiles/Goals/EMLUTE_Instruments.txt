// Master configuration management for Ember's Lute music mods.
// Spell container is added and removed on game enter and exit.
//REGION
IF
LevelGameplayStarted(_,_)
AND
DB_PartOfTheTeam(_Player)
AND
GetEquippedItem(_Player, "MusicalInstrument", _Instrument)
AND
QRY_IsEmbersLuteInstrument(_Instrument)
AND
HasSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 0)
THEN
AddSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 1, 1);
//
// Also added on equip
IF
Equipped(_Instrument, _Player)
AND
DB_PartOfTheTeam(_Player)
AND
QRY_IsEmbersLuteInstrument(_Instrument)
AND
HasSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 0)
THEN
AddSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 1, 1);
//
// And remove on unequip (with delay to only remove if necessary)
IF
Unequipped(_Instrument, _Player)
AND
DB_PartOfTheTeam(_Player)
AND
QRY_IsEmbersLuteInstrument(_Instrument)
AND
HasSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 1)
THEN
ObjectTimerCancel(_Player,"EmbersLute_ConfigSpellRemoveDelay");
ObjectTimerLaunch(_Player,"EmbersLute_ConfigSpellRemoveDelay",1500);
IF
ObjectTimerFinished((CHARACTER)_Player, "EmbersLute_ConfigSpellRemoveDelay")
AND
NOT QRY_HasEmbersLuteInstrument(_Player)
THEN
RemoveSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 1);

//
// Also add for varnish tag set
IF
TagSet((CHARACTER)_Player,(TAG)EMLUTE_INSTRUMENT_VARNISH_6dfa4afa-a78c-4240-98c8-3d6b615d8883)
AND
HasSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 0)
THEN
AddSpell(_Player, "Shout_ConfigManager_Cont_Emlute", 1, 1);
//END_REGION

///////////////////////////////////////////////////////////////////
// Instrument passive management events.
// Player equips a mod instrument.
// Add any passive toggles required by the instrument.
//REGION
IF
Equipped(_Instrument, _Character)
AND
DB_Players(_Character)
AND
DB_TagToPassiveMap_EmbersLute(_Tag, _PassiveToggle)
AND
IsTagged(_Instrument, _Tag, 1)
THEN
PROC_Emlute_AddPassive(_Character, _PassiveToggle);

// Any player unequips an Ember's Lute mod instrument.
// Clear up passive toggles no longer required.
IF
Unequipped(_Instrument, _Character)
AND
DB_Players(_Character)
AND
DB_TagToPassiveMap_EmbersLute(_Tag, _PassiveToggle)
AND
IsTagged(_Instrument, _Tag, 1)
THEN
PROC_Emlute_RemovePassiveToggle(_Character, _Instrument, _PassiveToggle);

// Any player unequips any Ember's Lute mod instrument, or a varnished instrument.
// Spike 1 turn of a varnish status overwrite to ensure it gets removed.
// Call the spell to dismiss all custom summons in the area.
IF
Unequipped(_Instrument, _Character)
AND
DB_Players(_Character)
AND
QRY_IsEmbersLuteInstrument(_Instrument)
THEN
ApplyStatus(_Instrument,"EMLUTE_VARNISH_INSTRUMENT_CLEAR",1.0);
UseSpell(_Character,"Shout_Dismiss_All_MusicalSpirits",_Character);

// Handle edge case for hot-swapping a varnished instrument with a custom instrument
// Need to throw in a short delay while the status clears up the passives
// ... and then check if we have to put them all back on again
IF
TagCleared(_Character,(TAG)EMLUTE_INSTRUMENT_VARNISH_6dfa4afa-a78c-4240-98c8-3d6b615d8883)
THEN
DebugText(_Character, "Varnish status removed from player");
RealtimeObjectTimerCancel(_Character,"EmbersLute_VarnishedRemoved");
RealtimeObjectTimerLaunch(_Character,"EmbersLute_VarnishedRemoved",1500);
IF
ObjectTimerFinished((CHARACTER)_Character, "EmbersLute_VarnishedRemoved")
AND
DB_Players(_Character)
AND
GetEquippedItem(_Character, "MusicalInstrument", _Instrument)
AND
DB_TagToPassiveMap_EmbersLute(_Tag, _PassiveToggle)
AND
IsTagged(_Instrument, _Tag, 1)
THEN
PROC_Emlute_AddPassive(_Character, _PassiveToggle);

PROC
PROC_Emlute_RemovePassiveToggle((CHARACTER)_Character, (ITEM)_Instrument, (STRING)_PassiveToggle)
AND
DB_PassiveTimer_EmbersLute(_, _PassiveToggle, _Delay)
AND
Concatenate("Remove check timer for: ", _PassiveToggle, _DebugMsg)
THEN
// THROW TIMER
DebugText(_Character,_DebugMsg);
RealtimeObjectTimerCancel(_Character,_PassiveToggle);
RealtimeObjectTimerLaunch(_Character,_PassiveToggle,_Delay);
// CATCH TIMER
// Only remove toggle if no longer needed.
IF
ObjectTimerFinished((CHARACTER)_Character, _PassiveToggle)
AND
DB_Players(_Character)
AND
DB_PassiveTimer_EmbersLute(_,_PassiveToggle,_)
AND
NOT QRY_HasEmbersLuteInstrument(_Character)
THEN
PROC_Emlute_RemovePassive(_Character, _PassiveToggle);
//
// This handles the instrument hot-swap cases.
IF
ObjectTimerFinished((CHARACTER)_Character, _PassiveToggle)
AND
DB_Players(_Character)
AND
DB_PassiveTimer_EmbersLute(_,_PassiveToggle,_)
AND
GetEquippedItem(_Character, "MusicalInstrument", _Instrument)
AND
NOT QRY_InstrumentHasPassiveToggle(_Instrument,_PassiveToggle)
THEN
PROC_Emlute_RemovePassive(_Character, _PassiveToggle);
//END_REGION

// End instrument passive management events

ClearTag(_Character,(TAG)MUSIC_VFX_VIOLIN_81dbb2dd-ffe9-4b6a-9159-2c32c889e023);
RealtimeObjectTimerLaunch(_Character,"ViolinBowPhotomodeHack_EmbersLute",1500);

// Catch the instrument hotswap case after a short pause, put the tag back
IF
ObjectTimerFinished((CHARACTER)_Character,"ViolinBowPhotomodeHack_EmbersLute")

IsTagged(_Instrument,(TAG)MUSIC_VFX_VIOLIN_81dbb2dd-ffe9-4b6a-9159-2c32c889e023,1)
THEN

//END_REGION

// Implementation of mechanic to grant a point of BardicInspiration on lute equip
// Avoids the farming of action resources via unequip and re-equip
//REGION
// When the lute is equipped, grant the charge only if we haven't already and the preference is enabled
IF
TagSet((CHARACTER)_Character,(TAG)EMLUTE_ENCHANTEDINSTRUMENT_HASPASSIVE_e7336d87-e1e4-4648-811f-ca2f4fde9c79)
AND
IsTagged(_Character,(TAG)EMLUTE_CONFIG_BARDIC_INSPIRATION_ON_b4bec563-e20b-40bb-a8e7-0d47ceb096a1,1)
AND
HasActiveStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION", 0)
AND
NOT DB_BardicInspirationBonusGranted_EmbersLute(_Character)
THEN
DB_BardicInspirationBonusGranted_EmbersLute(_Character);
ApplyStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION", -1.0);

// When a Bardic Inspiration resource is used, we remove the status and track the consumption
IF
TagSet((CHARACTER)_Character,(TAG)EMLUTE_ENCHANTEDINSTRUMENT_CONSUMED_6b32b3ee-3da3-4888-b18b-c90683a9bcfe)
AND
DB_BardicInspirationBonusGranted_EmbersLute(_Character)
AND
HasActiveStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION", 1)
THEN
DebugText(_Character,"Used Bardic Inspiration");
DB_BardicInspirationBonusConsumed_EmbersLute(_Character);
RemoveStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION");

// When unequipped, throw a timer to be sure the tag is still cleared after a second
// (To catch case where two instruments with the tag are swapped in inventory slot, rather than removed outright) 
IF
TagCleared((CHARACTER)_Character,(TAG)EMLUTE_ENCHANTEDINSTRUMENT_HASPASSIVE_e7336d87-e1e4-4648-811f-ca2f4fde9c79)
AND
HasActiveStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION", 1)
THEN
RealtimeObjectTimerLaunch(_Character,"EmbersLute_EnchantedInstrumentUnequipped_Timer",2200);

// If the charge is consumed, clear the status but retain the granted flag to prevent subsequent charges awarded
IF
ObjectTimerFinished((CHARACTER)_Character,"EmbersLute_EnchantedInstrumentUnequipped_Timer")
AND
IsTagged(_Character, (TAG)EMLUTE_ENCHANTEDINSTRUMENT_HASPASSIVE_e7336d87-e1e4-4648-811f-ca2f4fde9c79, 0)
AND
DB_BardicInspirationBonusConsumed_EmbersLute(_Character)
THEN
RemoveStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION");

// If the charge is not consumed, clear the status and the granted flag
IF
ObjectTimerFinished((CHARACTER)_Character,"EmbersLute_EnchantedInstrumentUnequipped_Timer")
AND
IsTagged(_Character, (TAG)EMLUTE_ENCHANTEDINSTRUMENT_HASPASSIVE_e7336d87-e1e4-4648-811f-ca2f4fde9c79, 0)
AND
NOT DB_BardicInspirationBonusConsumed_EmbersLute(_Character)
THEN
NOT DB_BardicInspirationBonusGranted_EmbersLute(_Character);
RemoveStatus(_Character, "EMLUTE_ENCHANTEDINSTRUMENT_BARDIC_INSPIRATION");
